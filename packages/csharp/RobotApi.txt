// Roboter-API für C# – wird mit dem Nutzerprogramm kompiliert; Programme binden sie selbst ein:
// `using klonk;` (+ `using static klonk.Klonk;` für Farben und Wait; Ausgabe wie gewohnt mit Console.WriteLine). Jede Figur ist ein Objekt:
// `var karl = new Klonk("Karl"); karl.Step();`. Englische Namen in C#-Schreibweise (Step, IsWall, PlaceBrick(Red) …).
// Die Aufrufe gehen über Delegaten, die der Host (Host.cs) im AppDomain hinterlegt, an die JS-Roboter-API im Worker.
// Namensraum `klonk` klein (wie das Modul in JS/Python/Java) – `Klonk` hieße wie die Klasse und verdeckte sie.
#nullable enable
namespace klonk
{
    public sealed class Klonk
    {
        private static readonly System.Func<string, string[], string> CallHost =
            (System.Func<string, string[], string>)System.AppDomain.CurrentDomain.GetData("klonk.call")!;
        private static readonly System.Func<int, bool> CheckpointHost =
            (System.Func<int, bool>)System.AppDomain.CurrentDomain.GetData("klonk.checkpoint")!;
        private static readonly System.Action<string, string> VariableHost =
            (System.Action<string, string>)System.AppDomain.CurrentDomain.GetData("klonk.variable")!;
        private static readonly System.Action PauseHost =
            (System.Action)System.AppDomain.CurrentDomain.GetData("klonk.pause")!;
        private static readonly System.Action<string> HeapHost =
            (System.Action<string>)System.AppDomain.CurrentDomain.GetData("klonk.heap")!;

        public const string Red = "red";
        public const string Blue = "blue";
        public const string Yellow = "yellow";
        public const string Green = "green";
        public const string Orange = "orange";
        public const string Purple = "purple";
        public const string White = "white";
        public const string Black = "black";

        private static string Call(string name, params string[] args) => CallHost(name, args);
        private static string N(double value) => "n:" + value.ToString(System.Globalization.CultureInfo.InvariantCulture);
        private static string S(string value) => "s:" + value;

        /// <summary>Name der Figur (wie im Welt-Editor)</summary>
        public string Name { get; }

        /// <summary>Die erste Figur der Welt</summary>
        public Klonk() => Name = Call("$robot");
        /// <summary>Die Figur mit diesem Namen</summary>
        public Klonk(string name) => Name = Call("$robot", S(name));
        /// <summary>Die n-te Figur (ab 1)</summary>
        public Klonk(int number) => Name = Call("$robot", N(number));

        /// <summary>Neue Figur: Aussehen "male", "female", "diverse" oder eigenes Design, optional auf Feld x|y (sonst neben der zuletzt genutzten)</summary>
        public static Klonk Create(string name, string? look = null, int x = -1, int y = -1) => new Klonk(Call("$create", S(name), S(look ?? ""), N(x), N(y)));

        /// <summary>Design des Projekts als JSON-Text (Lizenz) – z. B. mit Newtonsoft: JObject.Parse(Klonk.LoadDesign("ritter"))</summary>
        public static string LoadDesign(string name) => Call("loadDesign", S(name));
        /// <summary>Design (JSON-Text) ins Projekt speichern: Datei name.klonkdesign.json (Lizenz)</summary>
        public static void SaveDesign(string name, string json) => Call("saveDesign", S(name), S(json));

        /// <summary>Alle Figuren der Welt</summary>
        public static Klonk[] All()
        {
            var names = Call("$names");
            if (names.Length == 0) return System.Array.Empty<Klonk>();
            var parts = names.Split('\n');
            var result = new Klonk[parts.Length];
            for (var i = 0; i < parts.Length; i++) result[i] = new Klonk(parts[i]);
            return result;
        }

        public override string ToString() => "Klonk(\"" + Name + "\")";

        private string Do(string method, params string[] args)
        {
            var all = new string[args.Length + 1];
            all[0] = S(Name);
            args.CopyTo(all, 1);
            return Call(method, all);
        }

        public void Step(int count = 1) => Do("step", N(count));
        public void TurnLeft() => Do("turnLeft");
        public void TurnRight() => Do("turnRight");
        public void PlaceBrick(string color = Red) => Do("placeBrick", S(color));
        public void PickBrick() => Do("pickBrick");
        public void SetMark(string color = Yellow) => Do("setMark", S(color));
        public void RemoveMark() => Do("removeMark");
        public bool IsWall() => Do("isWall") == "true";
        public bool IsBrick() => Do("isBrick") == "true";
        public bool IsBrick(string color) => Do("isBrick", S(color)) == "true";
        public bool IsMark() => Do("isMark") == "true";
        public bool IsMark(string color) => Do("isMark", S(color)) == "true";
        public bool IsFacingNorth() => Do("isFacingNorth") == "true";
        public bool IsFacingEast() => Do("isFacingEast") == "true";
        public bool IsFacingSouth() => Do("isFacingSouth") == "true";
        public bool IsFacingWest() => Do("isFacingWest") == "true";
        public int BrickCount() => int.Parse(Do("brickCount"));
        public int PositionX() => int.Parse(Do("positionX"));
        public int PositionY() => int.Parse(Do("positionY"));

        public static void Wait(double seconds = 1) => Call("wait", N(seconds));

        // Datenbanken (Klasse Database): Anfrage und Antwort als JSON-Text
        internal static string CallDatabase(string request) => Call("$db", S(request));

        // Debugger (vom Instrumentierer eingefügt)
        public static bool __cp(int line)
        {
            var stop = CheckpointHost(line);
            if (stop) heapVariables.Clear();
            return stop;
        }
        public static void __pause()
        {
            HeapHost(Heap.Build(heapVariables));
            PauseHost();
        }
        public static void __v(string name, object? value)
        {
            VariableHost(name, Show(value, true));
            heapVariables.Add((name, value));
        }
        private static readonly System.Collections.Generic.List<(string, object?)> heapVariables = new();

        private static string Show(object? value, bool quote)
        {
            if (value is null) return "null";
            if (value is string s) return quote ? "\"" + s + "\"" : s;
            if (value is char c) return quote ? "'" + c + "'" : c.ToString();
            if (value is bool b) return b ? "true" : "false";
            if (value is System.IFormattable f) return f.ToString(null, System.Globalization.CultureInfo.InvariantCulture);
            if (value is System.Collections.IEnumerable items)
            {
                var parts = new System.Collections.Generic.List<string>();
                foreach (var item in items)
                {
                    if (parts.Count == 20) { parts.Add("…"); break; }
                    parts.Add(Show(item, true));
                }
                return "[" + string.Join(", ", parts) + "]";
            }
            return value.ToString() ?? "";
        }
    }

    /// <summary>Speicherabbild fürs Objektdiagramm (Reflexion): Objekte ab den Variablen, gleiche Objekte = gleiche ID</summary>
    internal static class Heap
    {
        private const int MaxObjects = 60;
        private const int MaxFields = 30;
        private const int MaxText = 60;
        private const System.Reflection.BindingFlags Fields = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.DeclaredOnly;

        internal static string Build(System.Collections.Generic.List<(string, object?)> variables)
        {
            var objects = new System.Collections.Generic.List<object>();
            var truncated = false;
            var json = new System.Text.StringBuilder("{\"variables\":[");

            string Prim(string text) => "{\"value\":" + Quote(text.Length > MaxText ? text.Substring(0, MaxText - 1) + "…" : text) + "}";
            string Value(object? v)
            {
                switch (v)
                {
                    case null: return Prim("null");
                    case string s: return Prim("\"" + s + "\"");
                    case char c: return Prim("'" + c + "'");
                    case bool b: return Prim(b ? "true" : "false");
                    case System.Enum e: return Prim(e.ToString());
                }
                var type = v.GetType();
                if (type.IsPrimitive || v is decimal || v is System.DateTime || v is System.TimeSpan || v is System.Guid)
                    return Prim(v is System.IFormattable f ? f.ToString(null, System.Globalization.CultureInfo.InvariantCulture) : v.ToString() ?? "");
                for (var i = 0; i < objects.Count; i++) if (ReferenceEquals(objects[i], v)) return "{\"ref\":" + (i + 1) + "}";
                if (objects.Count >= MaxObjects)
                {
                    truncated = true;
                    return Prim("…");
                }
                objects.Add(v);
                return "{\"ref\":" + objects.Count + "}";
            }

            for (var i = 0; i < variables.Count; i++)
            {
                if (i > 0) json.Append(',');
                json.Append('[').Append(Quote(variables[i].Item1)).Append(',').Append(Value(variables[i].Item2)).Append(']');
            }
            json.Append("],\"objects\":[");
            for (var i = 0; i < objects.Count; i++)
            {
                var o = objects[i];
                var fields = new System.Text.StringBuilder();
                var count = 0;
                var more = false;
                string? robot = null;
                void Field(string name, object? value)
                {
                    if (count >= MaxFields) { more = true; return; }
                    if (count++ > 0) fields.Append(',');
                    fields.Append('[').Append(Quote(name)).Append(',').Append(Value(value)).Append(']');
                }
                var type = o.GetType();
                var className = type.Name;
                var tick = className.IndexOf('`');
                if (tick > 0) className = className.Substring(0, tick);
                if (o is Klonk k)
                {
                    robot = k.Name;
                    Field("Name", k.Name);
                }
                else if (o is Database d)
                {
                    Field("Name", d.Name);
                    Field("Engine", d.Engine);
                }
                else if (o is System.Collections.IDictionary dict)
                {
                    foreach (System.Collections.DictionaryEntry entry in dict) Field("[" + (entry.Key is string s ? "\"" + s + "\"" : System.Convert.ToString(entry.Key, System.Globalization.CultureInfo.InvariantCulture)) + "]", entry.Value);
                }
                else if (o is System.Collections.IEnumerable items)
                {
                    if (type.IsArray) className = type.GetElementType()!.Name + "[]";
                    var n = 0;
                    foreach (var item in items) Field("[" + n++ + "]", item);
                }
                else
                {
                    // Attribute der Klasse und ihrer Oberklassen (Oberklasse zuerst); Auto-Properties ohne Hilfsnamen
                    var chain = new System.Collections.Generic.List<System.Type>();
                    for (var t = type; t != null && t != typeof(object) && t != typeof(System.ValueType); t = t.BaseType) chain.Insert(0, t);
                    foreach (var t in chain)
                    {
                        foreach (var field in t.GetFields(Fields))
                        {
                            var name = field.Name;
                            if (name.StartsWith("<") && name.EndsWith(">k__BackingField")) name = name.Substring(1, name.IndexOf('>') - 1);
                            else if (name.Contains('<')) continue;
                            Field(name, field.GetValue(o));
                        }
                    }
                }
                if (i > 0) json.Append(',');
                json.Append("{\"id\":").Append(i + 1).Append(",\"className\":").Append(Quote(className)).Append(",\"fields\":[").Append(fields).Append(']');
                if (robot != null) json.Append(",\"robot\":").Append(Quote(robot));
                if (more) json.Append(",\"more\":true");
                json.Append('}');
            }
            json.Append("],\"truncated\":").Append(truncated ? "true" : "false").Append('}');
            return json.ToString();
        }

        private static string Quote(string text)
        {
            var output = new System.Text.StringBuilder("\"");
            foreach (var c in text)
            {
                if (c == '"' || c == '\\') output.Append('\\').Append(c);
                else if (c == '\n') output.Append("\\n");
                else if (c < 32) output.Append(' ');
                else output.Append(c);
            }
            return output.Append('"').ToString();
        }
    }

    /// <summary>Fehler der Datenbank, z. B. ein Syntaxfehler im SQL</summary>
    public sealed class DatabaseException : System.Exception
    {
        public DatabaseException(string message) : base(message) { }
    }

    /// <summary>Eine Ergebniszeile von Database.Query: Spaltenname ⇒ Wert (long, double, string, bool oder null)</summary>
    public sealed class Row : System.Collections.Generic.Dictionary<string, object?>
    {
        public string? GetString(string column) => this[column] switch { null => null, string s => s, bool b => b ? "true" : "false", System.IFormattable f => f.ToString(null, System.Globalization.CultureInfo.InvariantCulture), var v => v.ToString() };
        public int GetInt(string column) => (int)GetLong(column);
        public long GetLong(string column) => this[column] switch { null => 0, long l => l, double d => (long)d, bool b => b ? 1 : 0, _ => long.Parse(GetString(column)!, System.Globalization.CultureInfo.InvariantCulture) };
        public double GetDouble(string column) => this[column] switch { null => 0, long l => l, double d => d, bool b => b ? 1 : 0, _ => double.Parse(GetString(column)!, System.Globalization.CultureInfo.InvariantCulture) };
        public bool GetBool(string column) => this[column] switch { bool b => b, long l => l != 0, double d => d != 0, _ => false };
        public bool IsNull(string column) => this[column] is null;
        public override string ToString() => "{ " + string.Join(", ", System.Linq.Enumerable.Select(this, p => p.Key + ": " + (p.Value is string s ? "\"" + s + "\"" : GetString(p.Key) ?? "null"))) + " }";
    }

    /// <summary>
    /// Datenbank des Projekts (Database.Open("schule") ⇒ Datei schule.sqlite.sql bzw. schule.postgres.sql) oder nur für
    /// diesen Lauf (Database.Memory("postgres")). Platzhalter: SQLite ?, PostgreSQL $1.
    /// </summary>
    public sealed class Database
    {
        private readonly int handle;
        /// <summary>Name der Datenbank (Dateiname ohne Endung bzw. "memory")</summary>
        public string Name { get; }
        /// <summary>"sqlite" oder "postgres"</summary>
        public string Engine { get; }

        private Database(Newtonsoft.Json.Linq.JObject info)
        {
            handle = (int)info["handle"]!;
            Name = (string)info["name"]!;
            Engine = (string)info["engine"]!;
        }

        private static Newtonsoft.Json.Linq.JToken Call(Newtonsoft.Json.Linq.JObject request)
        {
            var answer = Newtonsoft.Json.Linq.JToken.Parse(Klonk.CallDatabase(request.ToString(Newtonsoft.Json.Formatting.None)));
            if (answer is Newtonsoft.Json.Linq.JObject o && o["sqlError"] is { } error) throw new DatabaseException((string)error!);
            return answer;
        }

        /// <summary>Datenbank des Projekts öffnen (Dateiname ohne Endung)</summary>
        public static Database Open(string name) => new((Newtonsoft.Json.Linq.JObject)Call(new() { ["op"] = "open", ["name"] = name }));
        /// <summary>Neue, leere Datenbank nur für diesen Lauf: "sqlite" (Standard) oder "postgres"</summary>
        public static Database Memory(string engine = "sqlite") => new((Newtonsoft.Json.Linq.JObject)Call(new() { ["op"] = "memory", ["engine"] = engine }));

        private Newtonsoft.Json.Linq.JObject Run(string sql, object?[] args)
        {
            var values = new Newtonsoft.Json.Linq.JArray();
            foreach (var arg in args) values.Add(arg switch { null => Newtonsoft.Json.Linq.JValue.CreateNull(), System.DateTime d => new Newtonsoft.Json.Linq.JValue(d.ToString("o")), _ => Newtonsoft.Json.Linq.JToken.FromObject(arg) });
            return (Newtonsoft.Json.Linq.JObject)Call(new() { ["op"] = "query", ["handle"] = handle, ["sql"] = sql, ["params"] = values });
        }

        // Arme als object: sonst wäre JToken der gemeinsame Typ und Newtonsoft verpackte die Werte implizit wieder
        private static object? Value(Newtonsoft.Json.Linq.JToken? token) => token?.Type switch
        {
            null or Newtonsoft.Json.Linq.JTokenType.Null => null,
            Newtonsoft.Json.Linq.JTokenType.Integer => (object)(long)token,
            Newtonsoft.Json.Linq.JTokenType.Float => (object)(double)token,
            Newtonsoft.Json.Linq.JTokenType.Boolean => (object)(bool)token,
            Newtonsoft.Json.Linq.JTokenType.String => (object?)(string?)token,
            _ => (object)token,
        };

        /// <summary>Abfrage ⇒ Zeilen (Spalte ⇒ Wert)</summary>
        public System.Collections.Generic.List<Row> Query(string sql, params object?[] args)
        {
            var result = Run(sql, args);
            var columns = (Newtonsoft.Json.Linq.JArray)result["columns"]!;
            var list = new System.Collections.Generic.List<Row>();
            foreach (var values in (Newtonsoft.Json.Linq.JArray)result["rows"]!)
            {
                var row = new Row();
                for (var c = 0; c < columns.Count; c++) row[(string)columns[c]!] = Value(values[c]);
                list.Add(row);
            }
            return list;
        }

        /// <summary>Erste Zeile oder null</summary>
        public Row? QueryOne(string sql, params object?[] args)
        {
            var rows = Query(sql, args);
            return rows.Count > 0 ? rows[0] : null;
        }

        /// <summary>Ändernde Anweisung (INSERT, UPDATE, DELETE, CREATE …) ⇒ Zahl geänderter Zeilen</summary>
        public int Execute(string sql, params object?[] args) => Run(sql, args)["changes"] is { Type: Newtonsoft.Json.Linq.JTokenType.Integer } n ? (int)n : 0;

        /// <summary>Namen der Tabellen und Sichten</summary>
        public System.Collections.Generic.List<string> Tables()
        {
            var names = (Newtonsoft.Json.Linq.JArray)Call(new() { ["op"] = "tables", ["handle"] = handle });
            var list = new System.Collections.Generic.List<string>();
            foreach (var name in names) list.Add((string)name!);
            return list;
        }

        public override string ToString() => "Database(\"" + Name + "\", " + Engine + ")";
    }
}
